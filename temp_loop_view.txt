def parse_rgb(c, device=None, dtype=torch.float32):
    """
    臾몄옄??'r,g,b' ?먮뒗 (r,g,b) ?먮뒗 ?먯꽌瑜?0..1 踰붿쐞 [3] ?먯꽌濡??듭씪
    """
    if isinstance(c, str):
        try:
            r, g, b = [float(x) for x in c.split(",")]
        except Exception:
            r, g, b = 0.0, 255.0, 255.0
        t = torch.tensor([r, g, b], dtype=dtype, device=device)
    elif isinstance(c, torch.Tensor):
        t = c.to(dtype=dtype, device=device).flatten()[:3]
    else:  # list/tuple
        r, g, b = c
        t = torch.tensor([float(r), float(g), float(b)], dtype=dtype, device=device)

    # 255 ?ㅼ??쇱씠硫?1 ?ㅼ??쇰줈 ?뺢퇋??
    if t.max().item() > 1.0:
        t = t / 255.0
    return t.clamp_(0.0, 1.0)

def compute_energy_rgb(
    alpha_1chw: torch.Tensor,   # [1,1,H,W], 理쒖쥌 ?뚰뙆
    on_bg: bool,
    width: int,
    blur: float,
    color,                      # "r,g,b" | (r,g,b) | torch.Tensor([r,g,b])
    gain: float,
    device,
    dtype
) -> torch.Tensor:
    """
    RVM 理쒖쥌 ?뚰뙆???대? 寃쎄퀎瑜??뉕쾶 異붿텧 ???됱쓣 ?낇? 諛섑솚.
    諛섑솚: [3,H,W] RGB 0..1
    """
    # ?대? 寃쎄퀎: alpha - erode(alpha)
    edge = (alpha_1chw - erode_gpu(alpha_1chw, int(max(1, width)))).clamp(0, 1)
    # 諛곌꼍留?移좏븷吏 ?щ?
    mask = (1.0 - alpha_1chw) if on_bg else torch.ones_like(alpha_1chw)
    # 遺?쒕읇寃?
    if float(blur) > 0:
        edge = blur_alpha_gpu(edge, float(blur))
    # ?됱긽 + 寃뚯씤
    col = parse_rgb(color, device=device, dtype=dtype).view(3, 1, 1)
    energy = (edge * max(0.0, float(gain))) * mask   # [1,1,H,W]
    return (col * energy.squeeze(1)).clamp(0.0, 1.0) # [3,H,W]

def _compute_ds_ratio(h, w, target_long: int) -> float:
    return float(min(target_long / max(h, w), 1.0))

def _parse_fps(s):
    s = str(s)
    if "/" in s:
        a, b = s.split("/")
        return float(a) / float(b)
    return float(s)

def run_loop(
    *,
    # ?꾩닔 ?낆텧??
    grab,                   # .read(timeout)->(ok, frame)
    W: int, H: int,
    rvm,
    device: str = "cuda",
    fp16: bool = True,

    # RVM ?댁긽???섏씠??
    target_long: int = 768,
    target_fps_str: str = "60000/1001",
    force_fps_str: str = "60000/1001",
    ffmpeg_in: str = "decklink",

    # Depth (鍮꾪솢????臾댁떆)
    use_depth: bool = False,
    depth_model=None,
    depth_worker=None,
    depth_invert: bool = True,
    depth_bokeh: bool = False,
    blur_weak: float = 1.2,
    blur_strong: float = 4.0,

    alpha_unsharp: float = 0.35,

    alpha_ema: float = 0.8,
    ema_adaptive: bool = True,

    multi_mode: str = "all",

    # ===== ENERGY (湲곕낯 + Y-band ?쒗븳) =====
    energy: bool = False,

    energy_width: int = 6,

    energy_gain: float = 2.0,
    energy_on_bg: bool = True,
    energy_pulse: bool = False,
    energy_pulse_speed: float = 1.0,
    energy_pulse_min: float = 0.7,

    energy_pulse_shape: str = "sine",
    energy_color_source=None,
    energy_debug: bool = False,

    # Y-band ?듭뀡
    energy_y_band_mode: str = "off",      # "off" | "abs" | "bbox"
    energy_y_band_center: float = 50.0,   # 0..100 (??0, ?꾨옒=100)
    energy_y_band_width: float = 14.0,    # 諛뺤뒪 ??%)
    energy_y_band_sigma: float = 4.0,     # >0 ??媛?곗떆???뚰봽??留덉뒪??

    # --- Energy per-person ?꾨떖 ?몄옄 ---
    energy_per_person: bool = False,
    energy_colors: str = "255,0,0|0,255,0|0,128,255|255,255,0",
    energy_max_instances: int = 4,
    energy_instance_thr: float = 0.5,
    energy_min_area: int = 2000,
    energy_sort_by: str = "area",
    energy_count: int = 3,
    energy_blend: str = "screen",

    ripple_enabled: bool = False,
    ripple_count: int = 3,
    ripple_width: int = 2,
    ripple_spacing: int = 28,
    ripple_speed: float = 60.0,
    ripple_start: int = 8,
    ripple_color: str = "255,180,80",
    ripple_blend: str = "screen",
    ripple_on_bg: bool = True,
    ripple_gain: float = 1.0,

    # Pencil (colored pencil outline)
    pencil_enabled: bool = False,
    pencil_color: str = "255,128,0",
    pencil_color2: str | None = None,
    pencil_multicolor: int = 0,       # 0 ?⑥깋 / 1 ?????쇳빀
    pencil_width: int = 3,            # ??援듦린(px)
    pencil_soft: int = 1,             # ??紐⑥꽌由??뚰봽?몃떇(px)
    pencil_gain: float = 1.0,         # ?먯? 媛뺣룄 ?ㅼ???
    pencil_opacity: float = 0.8,      # 理쒖쥌 遺덊닾紐낅룄(0..1)
    pencil_thresh: float = 0.08,      # ?먯? 媛먯? ?꾧퀎(??쓣?섎줉 誘쇨컧)
    pencil_rough: float = 0.7,        # 吏덇컧 嫄곗튌湲?0..1)
    pencil_jitter: float = 0.8,       # ???⑤┝/援닿끝(0..1)
    pencil_dash: int = 0,             # 0 ?ㅼ꽑 / 1 ?먯꽑
    pencil_dash_len: int = 12,        # ?먯꽑 湲몄씠(px)
    pencil_gap_len: int = 8,          # ?먯꽑 媛꾧꺽(px)
    pencil_on_bg: bool = True,        # 諛곌꼍履쎌뿉留??곗꽑 ?곸슜
    pencil_on_fg: bool = True,        # ?쇱궗泥??ㅺ낸?먮룄 ?댁쭩 寃뱀튂湲?
    pencil_blend: str = "alpha",      # alpha | add | screen | overlay

    yolo=None, yolo_img: int = 640, yolo_conf: float = 0.5,

    # ?꾨━酉?異쒕젰
    sender=None,                # Obs/DeckLink Sender ?먮뒗 None
    no_preview: bool = False,
    preview_scale: float = 0.5,
    show_depth: bool = False,

    # 濡쒓렇
    show_fps: bool = True,
    log_detail: int = 1,
):
    """硫붿씤 猷⑦봽 (beam ?쒓굅 踰꾩쟾, Y-band ?쒗븳留??곸슜)"""

    # ?곹깭 蹂??
    base_long_hi = int(target_long)
    prev_ds = None
    rec = [None, None, None, None]
    alpha_prev = None
    prev_keep = None
    last_depth = None

    # CPU <-> GPU 踰꾪띁
    cpu_buf = torch.empty((H, W, 3), dtype=torch.uint8, pin_memory=True)

    # 猷⑦봽
    t0 = time.time(); frames = 0
    while True:
        loop_start = time.time()
        ok, frame = grab.read(timeout=1.0)
        if not ok:
            print("[WARN] ?꾨젅???쎄린 ?ㅽ뙣."); break

        # depth 鍮꾨룞湲?媛깆떊
        if use_depth and depth_worker is not None:
            depth_worker.submit(frame)
        else:
            last_depth = None

        # ?낅젰 ?먯꽌??(BGR->RGB)
        prep_t0 = time.time()
        frame_c = np.ascontiguousarray(frame)
        u8 = torch.from_numpy(frame_c)
        try:
            if torch.cuda.is_available(): u8 = u8.pin_memory()
        except: pass
        u8 = u8.to(device, non_blocking=True).permute(2,0,1)  # BGR->CHW
        rgb01 = u8[[2,1,0], ...].to(dtype=(torch.float16 if fp16 else torch.float32)) * (1.0/255.0)
        t_in = rgb01.unsqueeze(0)
        try: t_in = t_in.contiguous(memory_format=torch.channels_last)
        except: pass
        prep_t1 = time.time()

        # depth ?숆린 異붾줎(?듭뀡)
        if use_depth and depth_worker is not None and depth_worker.latest is not None:
            last_depth = depth_worker.latest
        elif use_depth and (depth_worker is None) and (depth_model is not None):
            d = depth_model.infer(frame)
            last_depth = (1.0 - d) if depth_invert else d

        # ===== RVM 異붾줎 =====
        infer_t0 = time.time()
        cur_long = base_long_hi
        ds = _compute_ds_ratio(frame.shape[0], frame.shape[1], cur_long)
        if (prev_ds is None) or (abs(ds - prev_ds) > 1e-6):
            rec = [None, None, None, None]
            alpha_prev = None
            prev_ds = ds

        with torch.inference_mode():
            if fp16 and isinstance(device, str) and device.startswith("cuda"):
                with torch.amp.autocast('cuda', dtype=torch.float16):
                    fgr, pha, *rec = rvm(t_in, *rec, downsample_ratio=ds)
            else:
                fgr, pha, *rec = rvm(t_in, *rec, downsample_ratio=ds)
        infer_t1 = time.time()

        # ===== ?뚰뙆 ?꾩쿂由?=====
        alpha = pha[0:1].clamp(0,1)
        if alpha_sigma > 0: alpha = blur_alpha_gpu(alpha, float(alpha_sigma))
        if alpha_unsharp > 0: alpha = unsharp_mask_alpha(alpha, amount=float(alpha_unsharp), radius=1.0)

        alpha_2d = alpha.squeeze(0).squeeze(0)
        keep = select_component(
            alpha_2d,
            last_depth if (use_depth and last_depth is not None and depth_invert) else
            (1.0 - last_depth if (use_depth and last_depth is not None) else None),
            mode=multi_mode, center_xy=(W//2, H//2)
        )
        alpha_2d = (alpha_2d * keep).clamp(0,1)
        alpha = alpha_2d.unsqueeze(0).unsqueeze(0)

        # ???湲됰? ??rec/EMA 由ъ뀑
        if prev_keep is None:
            prev_keep = keep
        else:
            inter = (keep * prev_keep).sum().item()
            union = torch.clamp(keep + prev_keep, 0, 1).sum().item()
            iou = (inter / union) if union > 0 else 0.0
            if iou < 0.5:
                alpha_prev = None
                rec = [None, None, None, None]
            prev_keep = keep

        # EMA
        alpha, alpha_prev, prev_motion_mag = ema_smooth_alpha(
            alpha, alpha_prev, base_ema=float(alpha_ema),
            adaptive=bool(ema_adaptive), loop_start_ts=loop_start
        )

        # ?뺥깭??
        if alpha_close > 0: alpha = close_gpu(alpha, int(alpha_close))
        if alpha_dilate > 0: alpha = dilate_gpu(alpha, int(alpha_dilate))

        # ?⑹꽦???먯꽌
        alpha2d = alpha.squeeze(0).clamp(0, 1)   # [1,H,W]
        fgr_t   = fgr[0].clamp(0, 1)             # [3,H,W] RGB
        base_bg = t_in[0].clamp(0, 1)            # [3,H,W] RGB
        alpha   = alpha.float(); alpha2d = alpha2d.float(); fgr_t = fgr_t.float(); base_bg = base_bg.float()

        # ===== 諛곌꼍(蹂댁?) =====
        bg_t = base_bg
        if use_depth and depth_bokeh and (last_depth is not None):
            depth_t = torch.from_numpy(last_depth).to(device=base_bg.device, dtype=base_bg.dtype)
            if depth_t.ndim == 2: depth_t = depth_t.unsqueeze(0)  # [1,H,W]
            depth_t = depth_t.clamp(0, 1)
            if depth_invert: depth_t = 1.0 - depth_t

            weak_blur   = gaussian_blur_gpu(base_bg, ks=11, sigma=float(blur_weak))
            strong_blur = gaussian_blur_gpu(base_bg, ks=21, sigma=float(blur_strong))
            bg_mask = (1.0 - alpha2d).clamp(0, 1)
            bokeh_mix = weak_blur * (1.0 - depth_t) + strong_blur * depth_t
            bg_t = base_bg * (1.0 - bg_mask) + bokeh_mix * bg_mask

        # ===== ?⑹꽦 =====
        comp_t = fgr_t * alpha2d + bg_t * (1.0 - alpha2d)

        # Ripple
        if bool(ripple_enabled):
            _dtype = comp_t.dtype
            comp_t = comp_t.float()
            comp_t = draw_moving_outlines(
                comp_rgb_3hw=comp_t,
                alpha_1chw=alpha,
                dilate_fn=dilate_gpu,
                t_seconds=time.time(),
                count=int(ripple_count),
                width_px=int(ripple_width),
                spacing_px=int(ripple_spacing),
                speed_px_s=float(ripple_speed),
                start_offset_px=int(ripple_start),
                color=str(ripple_color),
                blend=str(ripple_blend),
                on_bg_only=bool(ripple_on_bg),
                ripple_gain=float(ripple_gain),
            ).to(dtype=_dtype)

        # ?쒖뒳
        if bool(pencil_enabled):
            comp_t = draw_pencil_outline(
                comp_rgb_3hw = comp_t,
                alpha_1hw    = alpha.squeeze(0),  # [1,H,W]
                pencil       = 1,
                pencil_color = str(pencil_color),
                pencil_color2= (None if pencil_color2 in (None, "None", "") else str(pencil_color2)),
                pencil_multicolor = int(pencil_multicolor),
                pencil_width = int(pencil_width),
                pencil_soft  = int(pencil_soft),
                pencil_gain  = float(pencil_gain),
                pencil_opacity = float(pencil_opacity),
                pencil_thresh  = float(pencil_thresh),
                pencil_rough   = float(pencil_rough),
                pencil_jitter  = float(pencil_jitter),
                pencil_dash    = int(pencil_dash),
                pencil_dash_len= int(pencil_dash_len),
                pencil_gap_len = int(pencil_gap_len),
                pencil_on_bg   = bool(pencil_on_bg),
                pencil_on_fg   = bool(pencil_on_fg),
                pencil_blend   = str(pencil_blend),
            )

        # ---- YOLOv8 instance masks (person=0) ----
        instance_masks = None
        if yolo is not None:
            try:
                # BGR numpy ?꾨젅?꾩씠 ?꾩슂 ???대? 'frame' ?먮낯???덉쑝誘濡??ъ슜
                # classes=[0] ??COCO person
                res = yolo.predict(
                    source=frame, imgsz=int(yolo_img), conf=float(yolo_conf),
                    classes=[0], verbose=False, device=0 if (device=="cuda" or str(device).startswith("cuda")) else None
                )
                if res and len(res) > 0 and res[0].masks is not None:
                    m = res[0].masks  # ultralytics.Masks
                    # m.data: [N,h,w] torch(float, 0..1) on some device
                    mdata = m.data  # (N,h,w)
                    if mdata.ndim == 3:
                        mdata = mdata.unsqueeze(1)  # [N,1,h,w]

                    # ?먮낯 ?꾨젅???ш린濡?由ъ궗?댁쫰
                    N = mdata.shape[0]
                    mfull = F.interpolate(
                        mdata.float(), size=(H, W), mode="bilinear", align_corners=False
                    )  # [N,1,H,W], 0..1
                    # 媛꾨떒 ?꾧퀎 + 理쒖냼硫댁쟻 ?꾪꽣
                    inst = []
                    for i in range(N):
                        mi = (mfull[i] >= 0.5).float()  # [1,H,W]
                        area = mi.sum().item()
                        if area >= 1000:  # ?꾩슂??CLI濡?類????덉쓬
                            inst.append(mi)
                    if len(inst) > 0:
                        instance_masks = inst
                        if energy_debug:
                            print(f"[EnergyDebug][YOLO] masks={N}, kept={len(instance_masks)}")
                    else:
                        if energy_debug:
                            print(f"[EnergyDebug][YOLO] masks={N}, kept=0")

            except Exception as _e:
                if 'energy_debug' in locals() and energy_debug:
                    print("[YOLO] fail:", _e)
                instance_masks = None

        # Debug: when YOLO missing/no masks and per-person requested, estimate CC count
        if energy_debug:
            try:
                _pp = bool(energy_per_person)
            except Exception:
                _pp = False
            if _pp and (instance_masks is None):
                try:
                    from module.effects.energy_per_person import _connected_instances_from_alpha as _cc_inst
                    cc_list = _cc_inst(
                        alpha_1chw=alpha,
                        thr=float(energy_instance_thr if 'energy_instance_thr' in locals() else 0.5),
                        min_area_px=int(energy_min_area if 'energy_min_area' in locals() else 2000),
                        max_instances=int(energy_max_instances if 'energy_max_instances' in locals() else 4),
                        sort_by=str(energy_sort_by if 'energy_sort_by' in locals() else 'area'),
                    )
                    print(f"[EnergyDebug][CC] instances={len(cc_list)}")
                except Exception:
                    print("[EnergyDebug][CC] count unavailable (helper import or args missing)")

        # ===== ?먮꼫吏 (apply_energy: Y-band ?ы븿) =====

        try:
            from module.effects.energy_per_person import draw_energy_per_person
        except Exception:
            draw_energy_per_person = None

        if bool(energy):
            try: energy_per_person
            except NameError: energy_per_person = False
            try: energy_colors
            except NameError: energy_colors = "255,0,0|0,255,0|0,128,255|255,255,0"
            try: energy_max_instances
            except NameError: energy_max_instances = 4
            try: energy_instance_thr
            except NameError: energy_instance_thr = 0.5
            try: energy_min_area
            except NameError: energy_min_area = 2000
            try: energy_sort_by
            except NameError: energy_sort_by = "area"  # "area"|"x"|"y"
            try: energy_count
            except NameError: energy_count = 3
            try: energy_blend
            except NameError: energy_blend = "screen"  # "add"|"screen"|"normal"

            # ripple ?뚮씪誘명꽣瑜??대룞 ?쇱씤???ъ궗???놁쑝硫?湲곕낯)
            try: ripple_spacing
            except NameError: ripple_spacing = 18
            try: ripple_speed
            except NameError: ripple_speed = 80.0
            try: ripple_start
            except NameError: ripple_start = 2
            # 理쒖쥌 寃뚯씤(?꾩뒪 ??湲곗〈 洹몃?濡?
            eff_gain = float(energy_gain)
            if energy_pulse:
                tt = time.time()
                f = max(0.0, float(energy_pulse_speed))
                pulse = (0.5 * (math.sin(2.0*math.pi*f*tt) + 1.0)) if (energy_pulse_shape!="triangle") else \
                        (2.0 * (0.5 - abs((tt*f)%1.0 - 0.5)))
                k0, k1 = sorted([float(energy_pulse_min), float(energy_pulse_max)])
                eff_gain *= (k0 + (k1 - k0) * pulse)

            # ---- Lightweight tracking: order masks by stable track id ----
            try:
                label_ids = []
                masks_src = instance_masks
                if masks_src is None:
                    # build CC masks so we can still track
                    masks_src = connected_instances_from_alpha(
                        alpha_1chw=alpha,
                        thr=float(energy_instance_thr) if 'energy_instance_thr' in locals() else 0.5,
                        min_area_px=int(energy_min_area) if 'energy_min_area' in locals() else 2000,
                        max_instances=int(energy_max_instances) if 'energy_max_instances' in locals() else 4,
                        sort_by=str(energy_sort_by) if 'energy_sort_by' in locals() else 'area',
                    )
                if masks_src is not None and len(masks_src) > 0:
                    dets = build_detections_from_masks(masks_src)
                    assigns = tracker.update(dets)
                    # reorder instance_masks in place for rendering path
                    instance_masks = [masks_src[di] for (tid, di) in assigns]
                    label_ids = [tid for (tid, di) in assigns]
            except Exception:
                pass

            if bool(energy_per_person):
                if instance_masks is not None:  # YOLO ?깃났 ???몄뒪?댁뒪 留덉뒪??吏곸젒 ?ъ슜
                    comp_t = draw_energy_from_instance_masks(
                        comp_rgb_3hw=comp_t,
                        instance_masks_1hw=instance_masks,
                        alpha_1chw=alpha,
                        dilate_fn=dilate_gpu,
                        t_seconds=time.perf_counter(),
                        count=int(energy_count),
                        width_px=int(energy_width),
                        spacing_px=int(ripple_spacing),
                        speed_px_s=float(ripple_speed),
                        start_offset_px=int(ripple_start),
                        gain=float(eff_gain),
                        blend=str(energy_blend),
                        on_bg_only=bool(energy_on_bg),
                        palette_str=str(energy_colors),
                    )
                else:
                    # YOLO ?ㅽ뙣 ??湲곗〈 alpha connected-components 遺꾧린
                    comp_t = draw_energy_per_person(
                        comp_rgb_3hw=comp_t,
                        alpha_1chw=alpha,
                        dilate_fn=dilate_gpu,
                        t_seconds=time.perf_counter(),
                        count=int(energy_count),
                        width_px=int(energy_width),
                        spacing_px=int(ripple_spacing),
                        speed_px_s=float(ripple_speed),
                        start_offset_px=int(ripple_start),
                        gain=float(eff_gain),
                        blend=str(energy_blend),
